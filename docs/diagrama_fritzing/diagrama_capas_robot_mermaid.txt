    %% =========================================================================
    %% INTERCONEXIONES ENTRE CAPAS
    %% =========================================================================
    UI1 & UI2 & UI3 <--> B1 & B2 & B3
    B4 <--> B3
    AND1 <--> B1 & B2
    AND2 -. Extracción SQLite .-> B4

    B2 <== "Tramas JSON WebSocket / TCP (robot-s3-steps-v3)" ==> NET2
    NET2 <--> P1 & P2
    P3 -. Alerta Parada .-> P1

    P1 == "Consignas de Misión (distancia / ángulo / manual)" ==> SC1
    SC2 == "Telemetría de Estado (pose, ticks, gyro, error)" ==> P2

    SC1 <--> CTRL1 & CTRL2
    CTRL1 & CTRL2 --> CTRL3 --> CTRL4
    SC2 <--> KIN1 & KIN2
    KIN1 & KIN2 <--> IMU3 & ENC3

    IMU1 --> IMU2 --> IMU3
    ENC1 --> ENC2 --> ENC3

    HW_PCNT == "Pulsos Contados por Hardware" ==> ENC1
    CTRL4 == "Duty Cycle PWM" ==> HW_LEDC

    HW_LEDC == "PWM FL/FR/BL/BR (GPIO 4, 5, 6, 7, 15, 16, 17, 18)" ==> DRV
    TXS == "GPIO 10, 11, 12, 13 (A1..A4)" ==> HW_PCNT
    IMU1 == "Bus I²C (GPIO 8 SDA, GPIO 9 SCL)" ==> SC2

    DRV ==> MOT
    MOT -. Retroalimentación Óptica Ranurada .-> TXS
    BAT ==> DRV & CAP2 & FUSE
    GND_BUS --- TXS & DRV & BAT & CAP1 & CAP2
